## 五、爬虫模板

```python
# assets/crawler.py
"""
简易电商爬虫模板
用法: python crawler.py --category 耳机 --max 100
输出: assets/data/products.csv + assets/data/images/
"""

import requests
import csv
import os
import time
from bs4 import BeautifulSoup
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path

# 配置
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
}
OUTPUT_DIR = Path("assets/data")
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
IMAGE_DIR = OUTPUT_DIR / "images"
IMAGE_DIR.mkdir(exist_ok=True)

def fetch_product_list(category: str, max_pages: int = 5):
    """爬取商品列表（替换为实际目标网站）"""
    products = []
    for page in range(1, max_pages + 1):
        url = f"https://example.com/search?q={category}&page={page}"
        resp = requests.get(url, headers=HEADERS)
        soup = BeautifulSoup(resp.text, "html.parser")
        # TODO: 解析商品列表
        # for item in soup.select(".product-item"):
        #     products.append({
        #         "sku_id": f"SKU{len(products)+1:04d}",
        #         "title": item.select_one(".title").text,
        #         "price": float(item.select_one(".price").text),
        #         "image_url": item.select_one("img")["src"],
        #         ...
        #     })
        time.sleep(1)  # 礼貌延迟
    return products

def download_image(sku_id: str, url: str):
    """下载单张图片"""
    try:
        resp = requests.get(url, headers=HEADERS, timeout=10)
        ext = url.split(".")[-1].split("?")[0]
        path = IMAGE_DIR / f"{sku_id}.{ext}"
        with open(path, "wb") as f:
            f.write(resp.content)
        return str(path)
    except Exception as e:
        print(f"下载失败 {sku_id}: {e}")
        return None

def save_to_csv(products: list):
    """保存商品数据到 CSV"""
    path = OUTPUT_DIR / "products.csv"
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=[
            "sku_id", "title", "category", "brand", "color",
            "price", "attrs", "description", "image_url"
        ])
        writer.writeheader()
        writer.writerows(products)
    print(f"已保存 {len(products)} 条商品到 {path}")

def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--category", default="耳机")
    parser.add_argument("--max", type=int, default=100)
    args = parser.parse_args()

    products = fetch_product_list(args.category)
    products = products[:args.max]

    # 多线程下载图片
    with ThreadPoolExecutor(max_workers=10) as pool:
        for p in products:
            pool.submit(download_image, p["sku_id"], p["image_url"])

    save_to_csv(products)
    print(f"完成！共获取 {len(products)} 个商品")

if __name__ == "__main__":
    main()
```

## 八、注意事项

1. **爬虫遵守 robots.txt**，控制请求频率
2. **图片统一命名** `{sku_id}.jpg`，方便程序和脚本读取
3. **CSV 编码用 UTF-8**，避免中文乱码
4. **爬取的数据仅做比赛演示**，正式商用需授权
5. **图片先压缩再入库**，单张建议小于 500KB，节省 OSS 流量和 embedding 时间
